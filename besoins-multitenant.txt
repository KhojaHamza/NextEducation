import jwt from 'jsonwebtoken';
import { prisma } from '../config/prisma.js';
import { AppError } from '../utils/AppError.js';
import { asyncHandler } from '../utils/asyncHandler.js';

// Checks the token and loads the current user.
// Someone who still has a temporary password can only use the routes protected by
// authenticateAllowingTemporaryPassword (read own profile, change password).
const buildAuthenticate = ({ allowTemporaryPassword }) =>
  asyncHandler(async (req, res, next) => {
    const header = req.headers.authorization;
    if (!header || !header.startsWith('Bearer ')) {
      throw new AppError('Authentication required', 401);
    }

    let payload;
    try {
      payload = jwt.verify(header.split(' ')[1], process.env.JWT_SECRET);
    } catch {
      throw new AppError('Invalid or expired token', 401);
    }

    const user = await prisma.user.findFirst({
      where: { id: Number(payload.sub), deletedAt: null, isActive: true },
      select: {
        id: true, email: true, role: true, firstName: true, lastName: true, mustChangePassword: true,
      },
    });
    if (!user) throw new AppError('User no longer exists or is disabled', 401);

    if (user.mustChangePassword && !allowTemporaryPassword) {
      throw new AppError('You must change your temporary password first', 403);
    }

    req.user = user;
    next();
  });

export const authenticate = buildAuthenticate({ allowTemporaryPassword: false });
export const authenticateAllowingTemporaryPassword = buildAuthenticate({ allowTemporaryPassword: true });

// Restricts a route to the given roles: authorize('ADMIN', 'ACCOUNTANT')
export const authorize = (...roles) => (req, res, next) => {
  if (!req.user || !roles.includes(req.user.role)) {
    return next(new AppError('You do not have permission to do this', 403));
  }
  next();
};

import { Router } from 'express';
import { z } from 'zod';
import rateLimit from 'express-rate-limit';
import * as controller from '../controllers/auth.controller.js';
import { validate } from '../middlewares/validate.js';
import { authenticate, authenticateAllowingTemporaryPassword, authorize } from '../middlewares/auth.js';

const router = Router();

// Slows down brute-force attempts on the login route
const loginLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  limit: 10,
  message: { status: 'error', message: 'Too many login attempts, try again later' },
});

const loginSchema = z.object({
  email: z.string().email(),
  password: z.string().min(1),
});

const passwordSchema = z.object({
  currentPassword: z.string().min(1),
  newPassword: z
    .string()
    .min(8, 'At least 8 characters')
    .regex(/[A-Z]/, 'At least one uppercase letter')
    .regex(/[0-9]/, 'At least one number'),
});

router.post('/login', loginLimiter, validate(loginSchema), controller.login);
router.get('/me', authenticateAllowingTemporaryPassword, controller.me);
router.patch('/password', authenticateAllowingTemporaryPassword, validate(passwordSchema), controller.changePassword);

// TEMPORARY route to test RBAC, we remove it later
router.get('/admin-only', authenticate, authorize('ADMIN'), (req, res) => {
  res.json({ status: 'success', message: 'Welcome, admin' });
});

export default router;

import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { prisma } from '../config/prisma.js';
import { AppError } from '../utils/AppError.js';

const signToken = (user) =>
  jwt.sign({ sub: String(user.id), role: user.role }, process.env.JWT_SECRET, {
    expiresIn: process.env.JWT_EXPIRES_IN || '1h',
  });

export const login = async (email, password) => {
  const user = await prisma.user.findFirst({
    where: { email: email.toLowerCase(), deletedAt: null },
  });

  // Same message for unknown email and wrong password, to avoid leaking which emails exist
  if (!user || !user.isActive || !(await bcrypt.compare(password, user.passwordHash))) {
    throw new AppError('Invalid email or password', 401);
  }

  const { passwordHash, ...safeUser } = user;
  return { token: signToken(user), user: safeUser };
};

export const changePassword = async (userId, currentPassword, newPassword) => {
  const user = await prisma.user.findUnique({ where: { id: userId } });

  if (!(await bcrypt.compare(currentPassword, user.passwordHash))) {
    throw new AppError('Current password is incorrect', 400);
  }

  await prisma.user.update({
    where: { id: userId },
    data: { passwordHash: await bcrypt.hash(newPassword, 10),
        mustChangePassword: false,
     },
  });
};

import bcrypt from 'bcryptjs';
import crypto from 'crypto';
import { AppError } from '../utils/AppError.js';

export const userSelect = {
  id: true, email: true, firstName: true, lastName: true, phone: true,
  role: true, isActive: true, mustChangePassword: true, createdAt: true,
};

export const generateTempPassword = () => {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789';
  const body = Array.from({ length: 10 }, () => chars[crypto.randomInt(chars.length)]).join('');
  return `${body}${crypto.randomInt(10, 100)}`;
};

export const createUserAccount = async (tx, { email, firstName, lastName, phone, role }) => {
  const exists = await tx.user.findUnique({ where: { email } });
  if (exists) throw new AppError('This email is already used (possibly by an archived account)', 409);

  const tempPassword = generateTempPassword();
  const user = await tx.user.create({
    data: {
      email, firstName, lastName, phone, role,
      passwordHash: await bcrypt.hash(tempPassword, 10),
      mustChangePassword: true,
    },
    select: userSelect, // never returns or logs the hash
  });
  return { user, tempPassword };
};

import { prisma } from '../config/prisma.js';
import { AppError } from '../utils/AppError.js';
import { Decimal, ZERO, fmt } from '../utils/format.js';

const todayUtc = () => {
  const d = new Date();
  d.setUTCHours(0, 0, 0, 0);
  return d;
};

export const resolveYearId = async (db, schoolYearId) => {
  if (schoolYearId) return schoolYearId;
  const current = await db.schoolYear.findFirst({ where: { isCurrent: true, deletedAt: null } });
  if (!current) throw new AppError('No current school year: set one first', 400);
  return current.id;
};

// Applies the total paid to the installments, oldest due date first
export const allocate = (items, paid) => {
  const today = todayUtc();
  let remaining = new Decimal(paid);
  let due = ZERO();
  let overdueAmount = ZERO();

  const rows = items.map((item) => {
    const amount = new Decimal(item.amount);
    const applied = Decimal.min(amount, remaining);
    remaining = remaining.minus(applied);
    const outstanding = amount.minus(applied);
    const overdue = outstanding.gt(0) && item.dueDate < today;

    due = due.plus(amount);
    if (overdue) overdueAmount = overdueAmount.plus(outstanding);

    return {
      id: item.id, name: item.name, type: item.type, dueDate: item.dueDate,
      amount, paid: applied, outstanding, overdue,
      status: outstanding.isZero() ? 'PAID' : applied.isZero() ? 'UNPAID' : 'PARTIAL',
    };
  });

  return { due, paid: new Decimal(paid), balance: due.minus(paid), overdueAmount, items: rows };
};

export const serializeAccount = (a) => ({
  due: fmt(a.due),
  paid: fmt(a.paid),
  balance: fmt(a.balance),
  overdueAmount: fmt(a.overdueAmount),
  items: a.items.map((r) => ({
    ...r, amount: fmt(r.amount), paid: fmt(r.paid), outstanding: fmt(r.outstanding),
  })),
});

// upToPaymentId: balance as it was when that payment was recorded (used on receipts)
export const computeStudentAccount = async (db, studentId, schoolYearId, { upToPaymentId } = {}) => {
  const enrollment = await db.enrollment.findFirst({
    where: { studentId, schoolYearId, deletedAt: null },
    include: { schoolClass: true, schoolYear: true },
  });
  if (!enrollment) throw new AppError('Student is not enrolled for this school year', 404);

  const [items, agg] = await Promise.all([
    db.feeItem.findMany({
      where: {
        schoolYearId, deletedAt: null,
        OR: [{ level: null }, { level: enrollment.schoolClass.level }],
      },
      orderBy: [{ dueDate: 'asc' }, { id: 'asc' }],
    }),
    db.payment.aggregate({
      _sum: { amount: true },
      where: {
        studentId, schoolYearId, cancelledAt: null,
        ...(upToPaymentId && { id: { lte: upToPaymentId } }),
      },
    }),
  ]);

  return { enrollment, ...allocate(items, agg._sum.amount ?? ZERO()) };
};

// All students of a year in 3 queries (used for the unpaid list and reminders)
export const computeYearAccounts = async (db, schoolYearId, { classId } = {}) => {
  const [enrollments, items, paidGroups] = await Promise.all([
    db.enrollment.findMany({
      where: {
        schoolYearId, deletedAt: null, student: { deletedAt: null },
        ...(classId && { classId }),
      },
      include: { student: true, schoolClass: true },
    }),
    db.feeItem.findMany({
      where: { schoolYearId, deletedAt: null },
      orderBy: [{ dueDate: 'asc' }, { id: 'asc' }],
    }),
    db.payment.groupBy({
      by: ['studentId'],
      where: { schoolYearId, cancelledAt: null },
      _sum: { amount: true },
    }),
  ]);

  const paidBy = new Map(paidGroups.map((g) => [g.studentId, g._sum.amount ?? ZERO()]));

  return enrollments.map((e) => {
    const applicable = items.filter((i) => i.level === null || i.level === e.schoolClass.level);
    return {
      student: e.student,
      schoolClass: e.schoolClass,
      ...allocate(applicable, paidBy.get(e.studentId) ?? ZERO()),
    };
  });
};

export const sendEmailNotification = async ({ userId, email, type, title, body, data, link }) => {
  let status = 'SKIPPED';
  let error = null;
  let sentAt = null;

  try {
    const { sent } = await sendMail({ to: email, subject: title, text: link ? `${body}\n\n${link}` : body });
    if (sent) { status = 'SENT'; sentAt = new Date(); }
  } catch (err) {
    status = 'FAILED';
    error = String(err.message).slice(0, 300);
  }

  try {
    return await prisma.notification.create({
      data: { userId, type, title, body, data: data ?? undefined, channel: 'EMAIL', status, error, sentAt },
    });
  } catch (err) {
    console.error('Could not log email notification', err);
    return { status };
  }
};

import { z } from 'zod';
import { parse } from 'csv-parse/sync';
import { prisma } from '../config/prisma.js';
import { AppError } from '../utils/AppError.js';
import { allocateMatricules, createStudentRecord } from './students.service.js';

// ---------- Export ----------

// Prefixes risky cells so Excel never runs them as formulas (CSV injection)
const cell = (value) => {
  let s = value == null ? '' : String(value);
  if (/^[=+\-@\t\r]/.test(s)) s = `'${s}`;
  return /[;"\n\r]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
};

export const exportStudentsCsv = async (query) => {
  const where = { isCurrent: true, deletedAt: null };
  const year = query.schoolYearId
    ? await prisma.schoolYear.findFirst({ where: { id: Number(query.schoolYearId), deletedAt: null } })
    : await prisma.schoolYear.findFirst({ where });

  const students = await prisma.student.findMany({
    where: { deletedAt: null },
    orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
    include: {
      parents: { include: { parent: { include: { user: { select: { email: true } } } } } },
      enrollments: { where: { deletedAt: null, ...(year && { schoolYearId: year.id }) }, include: { schoolClass: true } },
    },
  });

  const header = ['matricule', 'lastName', 'firstName', 'birthDate', 'gender', 'class', 'parentEmails'];
  const lines = students.map((s) => {
    const cls = s.enrollments[0]?.schoolClass;
    return [
      s.matricule, s.lastName, s.firstName, s.birthDate.toISOString().slice(0, 10), s.gender,
      cls ? `${cls.level} ${cls.section}` : '',
      s.parents.map((p) => p.parent.user.email).join(' | '),
    ].map(cell).join(';');
  });

  return {
    filename: `students-${year?.label ?? 'all'}.csv`,
    // BOM + semicolon so Excel (French locale) opens it correctly
    content: `\uFEFF${[header.join(';'), ...lines].join('\r\n')}\r\n`,
  };
};

// ---------- Import ----------

const rowSchema = z.object({
  lastName: z.string().trim().min(1, 'lastName is required').max(60),
  firstName: z.string().trim().min(1, 'firstName is required').max(60),
  birthDate: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'Use YYYY-MM-DD'),
  gender: z.enum(['MALE', 'FEMALE']),
  class: z.string().trim().optional(),
  parentEmail: z.string().trim().toLowerCase().email().optional().or(z.literal('')),
});

const normalize = (raw) => {
  const g = String(raw.gender ?? '').trim().toUpperCase();
  return {
    lastName: raw.lastname ?? '',
    firstName: raw.firstname ?? '',
    birthDate: raw.birthdate ?? '',
    gender: { M: 'MALE', F: 'FEMALE' }[g] ?? g,
    class: raw.class ?? '',
    parentEmail: raw.parentemail ?? '',
  };
};

const dupKey = (last, first, date) => `${last}|${first}|${date}`.toLowerCase();

export const importStudentsCsv = async (buffer, userId) => {
  let rows;
  try {
    rows = parse(buffer, {
      columns: (h) => h.map((x) => x.trim().toLowerCase()),
      bom: true, skip_empty_lines: true, trim: true, delimiter: [',', ';'],
    });
  } catch (e) {
    throw new AppError(`Invalid CSV file: ${e.message}`, 400);
  }
  if (!rows.length) throw new AppError('The file has no data rows', 400);
  if (rows.length > 1000) throw new AppError('Maximum 1000 rows per import', 400);

  const year = await prisma.schoolYear.findFirst({ where: { isCurrent: true, deletedAt: null } });
  if (!year) throw new AppError('No current school year: set one first', 400);

  const [classes, counts, parents, existing] = await Promise.all([
    prisma.schoolClass.findMany({ where: { schoolYearId: year.id, deletedAt: null } }),
    prisma.enrollment.groupBy({
      by: ['classId'], where: { schoolYearId: year.id, deletedAt: null }, _count: { _all: true },
    }),
    prisma.parent.findMany({ where: { deletedAt: null }, include: { user: { select: { email: true } } } }),
    prisma.student.findMany({ where: { deletedAt: null }, select: { firstName: true, lastName: true, birthDate: true } }),
  ]);

  const classByKey = new Map(classes.map((c) => [`${c.level} ${c.section}`.toUpperCase(), c]));
  const free = new Map(classes.map((c) => [
    c.id, c.capacity - (counts.find((x) => x.classId === c.id)?._count._all ?? 0),
  ]));
  const parentByEmail = new Map(parents.map((p) => [p.user.email, p.id]));
  const seen = new Set(existing.map((s) => dupKey(s.lastName, s.firstName, s.birthDate.toISOString().slice(0, 10))));

  const errors = [];
  const valid = [];

  rows.forEach((raw, i) => {
    const line = i + 2; // line 1 is the header
    const fail = (field, message) => errors.push({ line, field, message });

    const parsed = rowSchema.safeParse(normalize(raw));
    if (!parsed.success) {
      parsed.error.issues.forEach((is) => fail(is.path.join('.'), is.message));
      return;
    }
    const r = parsed.data;

    const date = new Date(r.birthDate);
    if (Number.isNaN(date.getTime()) || date >= new Date()) return fail('birthDate', 'Invalid or future date');

    const key = dupKey(r.lastName, r.firstName, r.birthDate);
    if (seen.has(key)) return fail('lastName', 'Possible duplicate student');
    seen.add(key);

    let cls = null;
    if (r.class) {
      cls = classByKey.get(r.class.replace(/\s+/g, ' ').toUpperCase());
      if (!cls) return fail('class', `Unknown class "${r.class}" for ${year.label}`);
      if (free.get(cls.id) <= 0) return fail('class', `Class ${r.class} is full`);
      free.set(cls.id, free.get(cls.id) - 1);
    }

    let parentId = null;
    if (r.parentEmail) {
      parentId = parentByEmail.get(r.parentEmail);
      if (!parentId) return fail('parentEmail', `No parent with email ${r.parentEmail}`);
    }

    valid.push({ r, date, cls, parentId });
  });

  if (errors.length) {
    throw new AppError(
      `Import rejected: ${errors.length} problem(s) found, nothing was imported`, 400, errors.slice(0, 50),
    );
  }

  await prisma.$transaction(
    async (tx) => {
      const matricules = await allocateMatricules(tx, year.id, valid.length);
      for (const [i, v] of valid.entries()) {
        await createStudentRecord(tx, {
          data: {
            firstName: v.r.firstName, lastName: v.r.lastName, birthDate: v.date, gender: v.r.gender,
            entryYearId: year.id, classId: v.cls?.id,
            parents: v.parentId ? [{ parentId: v.parentId, relationship: 'guardian' }] : [],
          },
          matricule: matricules[i],
          userId,
        });
      }
    },
    { timeout: 60000, maxWait: 10000 }, // the default 5 s is too short for large files
  );

  return { imported: valid.length };
};
