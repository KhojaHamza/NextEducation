Quand on essaie de se connecter (POST /api/auth/login) :
1. Le frontend affiche "Connexion impossible".
2. PowerShell renvoie soit 400 (Zod), soit 500 (Internal server error).
3. Le terminal affiche 429 après plusieurs essais (rate limit).

--- CE QUE J'AI FAIT ---
J'ai appliqué les migrations (Phase 3) et le code d'isolation (Phase 4).
Le backend a été redémarré.

import { AsyncLocalStorage } from 'node:async_hooks';
import { PrismaClient } from '@prisma/client';

// Per-request tenant context. Populated by `authenticate` (middlewares/auth.js).
export const schoolContext = new AsyncLocalStorage();

// Models that are NOT tenant-scoped.
const UNSCOPED_MODELS = new Set(['School']);

// Operations that require a unique `where` and cannot be injected with `schoolId`.
// We reject them so any callsite that would leak across schools fails loudly.
const FORBIDDEN_OPS = new Set(['findUnique', 'findUniqueOrThrow', 'upsert']);

// Operations that accept a `where` clause.
const WHERE_OPS = new Set([
  'findFirst', 'findFirstOrThrow', 'findMany',
  'update', 'updateMany', 'delete', 'deleteMany',
  'aggregate', 'count', 'groupBy',
]);

const rawClient = new PrismaClient();

export const prisma = rawClient.$extends({
  name: 'schoolIsolation',
  query: {
    $allModels: {
      async $allOperations({ model, operation, args, query }) {
        if (!model || UNSCOPED_MODELS.has(model)) {
          return query(args);
        }

        if (FORBIDDEN_OPS.has(operation)) {
          throw new Error(
            `[tenancy] ${model}.${operation} is forbidden: use findFirst with a school filter ` +
            `(schoolId is injected automatically by the extension).`,
          );
        }

        const ctx = schoolContext.getStore();
        if (!ctx) {
          throw new Error(
            `[tenancy] Refusing ${model}.${operation}: no request context. ` +
            `Wrap the call in schoolContext.run({ schoolId, ... }, ...) or use platformPrisma.`,
          );
        }

        // SUPER_ADMIN has no school: must use platformPrisma for any scoped model.
        if (ctx.role === 'SUPER_ADMIN' && ctx.schoolId === null) {
          throw new Error(
            `[tenancy] SUPER_ADMIN cannot access ${model}.${operation} through the scoped client. ` +
            `Use platformPrisma for platform-level operations.`,
          );
        }

        const schoolId = ctx.schoolId;
        if (!schoolId) {
          throw new Error(
            `[tenancy] Refusing ${model}.${operation}: no schoolId in request context.`,
          );
        }

        const scoped = { ...(args ?? {}) };

        if (WHERE_OPS.has(operation)) {
          scoped.where = { ...(scoped.where ?? {}), schoolId };
        }
        if (operation === 'create') {
          scoped.data = { ...(scoped.data ?? {}), schoolId };
        }
        if (operation === 'createMany') {
          scoped.data = (scoped.data ?? []).map((row) => ({ ...row, schoolId }));
        }

        return query(scoped);
      },
    },
  },
});

// Raw client that bypasses tenant isolation. Only for:
//   - platform login (SUPER_ADMIN)
//   - changePassword (userId comes from the verified JWT)
//   - seed scripts
// Any other use is a bug and must be flagged in code review.
export const platformPrisma = rawClient;

import jwt from 'jsonwebtoken';
import { prisma, platformPrisma, schoolContext } from '../config/prisma.js';
import { AppError } from '../utils/AppError.js';
import { asyncHandler } from '../utils/asyncHandler.js';

const buildAuthenticate = ({ allowTemporaryPassword }) =>
  asyncHandler(async (req, res, next) => {
    const header = req.headers.authorization;
    if (!header || !header.startsWith('Bearer ')) {
      throw new AppError('Authentication required', 401);
    }

    let payload;
    try {
      payload = jwt.verify(header.split(' ')[1], process.env.JWT_SECRET);
    } catch {
      throw new AppError('Invalid or expired token', 401);
    }

    // We don't know the school yet: use the platform client for this one lookup.
    const user = await platformPrisma.user.findFirst({
      where: { id: Number(payload.sub), deletedAt: null, isActive: true },
      select: {
        id: true, email: true, role: true, firstName: true, lastName: true,
        mustChangePassword: true, schoolId: true,
        school: { select: { status: true } },
      },
    });
    if (!user) throw new AppError('User no longer exists or is disabled', 401);

    // ---- SUPER_ADMIN: no school, no tenant context ----
    if (user.role === 'SUPER_ADMIN') {
      if (user.schoolId) throw new AppError('SUPER_ADMIN must not belong to a school', 500);
      return schoolContext.run(
        { userId: user.id, role: 'SUPER_ADMIN', schoolId: null },
        () => {
          req.user = {
            id: user.id, email: user.email, role: 'SUPER_ADMIN',
            firstName: user.firstName, lastName: user.lastName,
            mustChangePassword: user.mustChangePassword, schoolId: null,
          };
          next();
        },
      );
    }

    // ---- Regular users: must have an active school ----
    if (!user.schoolId) throw new AppError('User has no school', 403);
    if (!user.school || user.school.status !== 'ACTIVE') {
      throw new AppError('This school is not active', 403);
    }
    if (user.mustChangePassword && !allowTemporaryPassword) {
      throw new AppError('You must change your temporary password first', 403);
    }

    return schoolContext.run(
      { userId: user.id, schoolId: user.schoolId, role: user.role },
      () => {
        req.user = {
          id: user.id, email: user.email, role: user.role,
          firstName: user.firstName, lastName: user.lastName,
          mustChangePassword: user.mustChangePassword,
          schoolId: user.schoolId,
        };
        next();
      },
    );
  });

export const authenticate = buildAuthenticate({ allowTemporaryPassword: false });

export const validate = (schema) => (req, res, next) => {
  const result = schema.safeParse(req.body);
  if (!result.success) {
    const details = result.error.issues.map((i) => ({
      field: i.path.join('.'),
      message: i.message,
    }));
    return next(new AppError('Validation failed', 400, details));
  }
  req.body = result.data;
  next();
};

const fromPrisma = (err) => {
  if (err?.name !== 'PrismaClientKnownRequestError') return null;
  switch (err.code) {
    case 'P2002':
      return new AppError('A record with the same unique values already exists (it may be archived)', 409);
    case 'P2003':
      return new AppError('A referenced record does not exist', 400);
    case 'P2025':
      return new AppError('Record not found', 404);
    default:
      return null;
  }
};
    
// Translates upload errors (file too large, etc.)
const fromMulter = (err) =>
  err?.name === 'MulterError'
    ? new AppError(err.code === 'LIMIT_FILE_SIZE' ? 'File too large' : err.message, 400)
    : null;

export const errorHandler = (err, req, res, next) => {
  const error = fromMulter(err) || fromPrisma(err) || err;
  const statusCode = error.statusCode || 500;

  if (statusCode === 500) console.error(err);

  res.status(statusCode).json({
    status: 'error',
    message: statusCode === 500 ? 'Internal server error' : error.message,
    ...(error.details && { details: error.details }),
  });
};

import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { prisma, platformPrisma } from '../config/prisma.js';
import { AppError } from '../utils/AppError.js';

const signToken = (user) =>
  jwt.sign({ sub: String(user.id), role: user.role }, process.env.JWT_SECRET, {
    expiresIn: process.env.JWT_EXPIRES_IN || '1h',
  });

// School user login. Same error message for unknown school / email / password.
export const login = async (schoolCode, email, password) => {
  const school = await platformPrisma.school.findFirst({
    where: { code: schoolCode },
    select: { id: true, status: true, name: true, currency: true },
  });
  
  const fail = () => new AppError('Invalid credentials', 401);
  if (!school || school.status !== 'ACTIVE') throw fail();

  // On utilise platformPrisma pour chercher l'utilisateur avec son schoolId
  const user = await platformPrisma.user.findFirst({
    where: { email: email.toLowerCase(), schoolId: school.id, deletedAt: null },
  });

  if (!user || !user.isActive || !(await bcrypt.compare(password, user.passwordHash))) {
    throw fail();
  }

  const { passwordHash, ...safeUser } = user;
  return {
    token: signToken(user),
    user: safeUser,
    school: { id: school.id, code: schoolCode, name: school.name, currency: school.currency },
  };
};

export const platformLogin = async (email, password) => {
  const user = await platformPrisma.user.findFirst({
    where: { email: email.toLowerCase(), role: 'SUPER_ADMIN', deletedAt: null },
  });
  if (!user || !user.isActive || !(await bcrypt.compare(password, user.passwordHash))) {
    throw new AppError('Invalid credentials', 401);
  }
  const { passwordHash, ...safeUser } = user;
  return { token: signToken(user), user: safeUser };
};

export const changePassword = async (userId, currentPassword, newPassword) => {
  const user = await platformPrisma.user.findFirst({ where: { id: userId, deletedAt: null } });
  if (!user) throw new AppError('User not found', 401);
  if (!(await bcrypt.compare(currentPassword, user.passwordHash))) {
    throw new AppError('Current password is incorrect', 400);
  }
  await platformPrisma.user.update({
    where: { id: userId },
    data: {
      passwordHash: await bcrypt.hash(newPassword, 10),
      mustChangePassword: false,
    },
  });
};

===== frontend\src\features\auth\AuthContext.jsx =====
import { createContext, useCallback, useEffect, useMemo, useState } from 'react';
import { api, setSessionExpiredHandler, tokenStore } from '../../api/client.js';
import { queryClient } from '../../app/queryClient.js';

export const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [status, setStatus] = useState(() => (tokenStore.get() ? 'loading' : 'anonymous'));

  useEffect(() => {
    setSessionExpiredHandler(() => {
      setUser(null);
      setStatus('anonymous');
    });
  }, []);

  useEffect(() => {
    if (!tokenStore.get()) return undefined;
    const controller = new AbortController();

    api('/auth/me', { signal: controller.signal })
      .then((data) => {
        setUser(data.user);
        setStatus('authenticated');
      })
      .catch((err) => {
        if (err.name === 'AbortError') return;
        setUser(null);
        setStatus('anonymous');
      });

    return () => controller.abort();
  }, []);

  const login = useCallback(async (schoolCode, email, password) => {
    const data = await api('/auth/login', { method: 'POST', body: { schoolCode, email, password } });
    tokenStore.set(data.token);
    setUser(data.user);
    setStatus('authenticated');
    return data.user;
  }, []);

  const logout = useCallback(() => {
    tokenStore.clear();
    queryClient.clear();
    setUser(null);
    setStatus('anonymous');
  }, []);

  const markPasswordChanged = useCallback(() => {
    setUser((current) => (current ? { ...current, mustChangePassword: false } : current));
  }, []);

  const value = useMemo(
    () => ({ user, status, login, logout, markPasswordChanged }),
    [user, status, login, logout, markPasswordChanged],
  );
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}
